cmake_minimum_required(VERSION 3.25)
project(VegasCraft VERSION 0.2.2 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "The New Vegas plugin and native test host must target 32-bit Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(vegas_bridge STATIC native/Bridge.cpp)
target_include_directories(vegas_bridge PUBLIC native protocol)
target_compile_definitions(vegas_bridge PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
if(MSVC)
  target_compile_options(vegas_bridge PUBLIC /W4)
else()
  target_compile_options(vegas_bridge PUBLIC -Wall -Wextra -Wpedantic)
  add_link_options(-static)
endif()
# The xNVSE plugin is a small loader (bridge, log, Minecraft launcher) plus a core it can reload
# while New Vegas runs (native/CoreAbi.h).
add_library(VegasCraft SHARED native/Plugin.cpp native/Log.cpp native/Launcher.cpp native/exports.def)
target_link_libraries(VegasCraft PRIVATE vegas_bridge ole32 oleaut32 uuid shell32)
set_target_properties(VegasCraft PROPERTIES PREFIX "")
add_library(VegasCraftCore SHARED native/Core.cpp native/Session.cpp native/Collision.cpp native/Input.cpp
  native/Overlay.cpp native/WorldRender.cpp native/Log.cpp native/Probe.cpp native/Rtti.cpp native/Combat.cpp native/NpcBlocks.cpp native/Dig.cpp
  native/Camera.cpp native/Hud.cpp native/BlockLights.cpp native/DigMesh.cpp native/DigPhysics.cpp native/Items.cpp native/Inventory.cpp native/PipData.cpp native/Arms.cpp)
target_link_libraries(VegasCraftCore PRIVATE vegas_bridge d3d9 dinput8 dxguid ole32 oleaut32 uuid shell32)
set_target_properties(VegasCraftCore PROPERTIES PREFIX "")
add_dependencies(VegasCraft VegasCraftCore)
add_executable(bridge_host tests/bridge_host.cpp)
target_link_libraries(bridge_host PRIVATE vegas_bridge)
add_executable(homography_tests tests/homography_tests.cpp)
target_include_directories(homography_tests PRIVATE native)
add_executable(plugin_tests tests/plugin_tests.cpp)
target_link_libraries(plugin_tests PRIVATE vegas_bridge)
add_executable(render_preview tests/render_preview.cpp native/MotionPreview.cpp)
target_link_libraries(render_preview PRIVATE vegas_bridge d3d9)
add_executable(gameplay_tests tests/gameplay_tests.cpp native/Collision.cpp native/Dig.cpp native/DigMesh.cpp native/DigPhysics.cpp native/Camera.cpp native/BlockLights.cpp native/Log.cpp native/Rtti.cpp)
target_link_libraries(gameplay_tests PRIVATE vegas_bridge)
target_compile_definitions(gameplay_tests PRIVATE VEGAS_TESTING)
add_executable(pip_data_tests tests/pip_data_tests.cpp native/Inventory.cpp native/PipData.cpp native/Items.cpp native/Log.cpp)
target_link_libraries(pip_data_tests PRIVATE vegas_bridge)
# The fixture must emulate FalloutNV's fixed address range. Keep only the test executable
# at a known base so ASLR cannot randomly place its own image over the synthetic singleton.
if(MSVC)
  target_link_options(plugin_tests PRIVATE /BASE:0x400000 /DYNAMICBASE:NO)
  target_link_options(gameplay_tests PRIVATE /BASE:0x4000000 /DYNAMICBASE:NO)
  target_link_options(pip_data_tests PRIVATE /BASE:0x5000000 /DYNAMICBASE:NO)
else()
  target_link_options(plugin_tests PRIVATE -Wl,--image-base,0x400000,--disable-dynamicbase)
  target_link_options(gameplay_tests PRIVATE -Wl,--image-base,0x4000000,--disable-dynamicbase)
  target_link_options(pip_data_tests PRIVATE -Wl,--image-base,0x5000000,--disable-dynamicbase)
endif()
enable_testing()
add_test(NAME plugin_lifecycle COMMAND plugin_tests $<TARGET_FILE:VegasCraft>)
add_test(NAME homography COMMAND homography_tests)
add_test(NAME render_fixture COMMAND render_preview)
add_test(NAME gameplay_fixture COMMAND gameplay_tests)
add_test(NAME pipboy_data COMMAND pip_data_tests)
install(TARGETS VegasCraft RUNTIME DESTINATION Data/NVSE/Plugins)
install(TARGETS VegasCraftCore RUNTIME DESTINATION Data/NVSE/Plugins/VegasCraft)
install(FILES config/VegasCraft.ini DESTINATION Data/NVSE/Plugins)
